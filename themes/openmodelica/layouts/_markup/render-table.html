{{- /* Markdown tables get the class "table", like the HTML tables in the pages, so that table.table in the stylesheet styles both. On pages with "mergeEmptyCells: true" (the workshop programmes, through "cascade" in their _index.md), empty cells at the end of a row are merged into the last non-empty cell, for example for a lunch break that has no speaker and no talk. */ -}} {{- $merge := .Page.Params.mergeEmptyCells -}} {{- range .THead }} {{- range . }} {{- end }} {{- end }} {{- range .TBody }} {{- $row := . }} {{- $last := sub (len $row) 1 }} {{- if $merge }} {{- $last = 0 }} {{- range $i, $cell := $row }} {{- if ne (trim (string $cell.Text) " ") "" }}{{ $last = $i }}{{ end }} {{- end }} {{- end }} {{- $span := sub (len $row) $last }} {{- range $i, $cell := $row }} {{- if lt $i $last }} {{- else if and (eq $i $last) (gt $span 1) }} {{- else if eq $i $last }} {{- end }} {{- end }} {{- end }}
